\AtBeginDocument{\hypersetup{colorlinks=true, allcolors=black}}

% stretch must match linestretch in _quarto.yml
\usepackage{caption}
\captionsetup{format=plain, labelsep=newline, labelfont=bf, textfont=it, justification=raggedright, singlelinecheck=false, font={stretch=2.0}}
\setlength{\abovecaptionskip}{4pt}

\usepackage{scrlayer-scrpage}
\clearpairofpagestyles
\ohead{\pagemark}
\setkomafont{disposition}{\normalfont\bfseries}

\setlength{\parindent}{0.5in}
\usepackage{indentfirst}

\usepackage{floatrow}
\floatsetup[figure]{capposition=top}
\usepackage{float}
\makeatletter
\def\fps@figure{H}
\makeatother

% afterskip 1pt rather than 0pt keeps the headings display-style
\RedeclareSectionCommands[
  beforeskip=0pt,
  afterskip=1pt
]{chapter,section,subsection,subsubsection,paragraph}

% Heading fonts stay body-size: larger fonts re-inflate the glue above
% each heading line and break the uniform double-spaced grid.
\setkomafont{chapter}{\normalsize}
\setkomafont{section}{\normalsize}
\setkomafont{subsection}{\normalsize}
\setkomafont{subsubsection}{\normalsize}
\setkomafont{paragraph}{\normalsize}
\setkomafont{subparagraph}{\normalsize}

% enumitem, because KOMA's \@list levels reset raw \topsep/\itemsep settings.
\setlength{\parskip}{0pt}
\usepackage{enumitem}
\setlist{nosep}

% One body line: set at begin document so \baselineskip carries the line
% stretch; +5.2pt compensates the float box join.
\AtBeginDocument{\setlength{\intextsep}{\dimexpr\baselineskip+5.2pt\relax}}

\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\footnotesize}
\AtBeginEnvironment{tabular}{\footnotesize}
\setlength{\LTleft}{0pt}
\setlength{\LTright}{0pt}
\setlength{\LTcapwidth}{\textwidth}
% Calibrated so a mid-page caption sits one body line below the callout
% (at page top TeX discards the skip).
\setlength{\LTpre}{7.6pt}
\setlength{\LTpost}{0pt}

% No autonumber where the title already says "Chapter N" (chapters 1 and 3).
% Tests \thechapter, not the counter, so the appendix (counter 1, rendered
% "A") keeps its number.
\renewcommand*{\raggedchapter}{\centering}
\makeatletter
\renewcommand*{\chapterformat}{%
  \edef\reserved@a{\thechapter}%
  \def\reserved@b{1}%
  \ifx\reserved@a\reserved@b\else
    \ifnum\value{chapter}=3\relax\else
      \thechapter\enspace
    \fi
  \fi}
\makeatother

% Overrides KOMA's default, which ends the number in a trailing dot.
\renewcommand*{\sectionformat}{\thesection\enskip}
\renewcommand*{\subsectionformat}{\thesubsection\enskip}
\renewcommand*{\subsubsectionformat}{\thesubsubsection\enskip}
\renewcommand*{\paragraphformat}{\theparagraph\enskip}
\renewcommand*{\subparagraphformat}{\thesubparagraph\enskip}
